::warning
Next is the one framework without a packaged integration. It works, but you write the wiring
yourself. A `@quello/next` package is on the roadmap.
::

## Why

Next builds on webpack or turbopack, but neither is reachable the usual way:

- it runs **its own dev server** rather than `webpack-dev-server`, so there is no `setupMiddlewares`
  hook to add the picks endpoint to;
- it renders HTML with **its own renderer** rather than `html-webpack-plugin`, so there is no
  generated document to add the script tag to.

Both halves therefore go the Next way: a client component that starts the runtime, and a route
handler that stores the picks.

## Install

```bash [Terminal]
pnpm add -D @quello/core
```

## The client component

```tsx [components/QuelloDev.tsx]
'use client'

import { useEffect } from 'react'

export function QuelloDev() {
  useEffect(() => {
    if (process.env.NODE_ENV !== 'development') return
    let stop: (() => void) | undefined
    void import('@quello/core').then(({ createQuello }) => {
      const instance = createQuello({ endpoint: '/api/quello/picks' })
      stop = () => instance.destroy()
    })
    return () => stop?.()
  }, [])

  return null
}
```

Render it once, in your root layout:

```tsx [app/layout.tsx]
import { QuelloDev } from '../components/QuelloDev'

export default function RootLayout({ children }: { children: React.ReactNode }) {
  return (
    <html lang="en">
      <body>
        {children}
        <QuelloDev />
      </body>
    </html>
  )
}
```

## The route handler

```ts [app/api/quello/picks/route.ts]
import { mkdir, readFile, writeFile } from 'node:fs/promises'
import { dirname, resolve } from 'node:path'
import { NextResponse } from 'next/server'
import type { QuelloPick, QuelloPicksFile } from '@quello/core'

const FILE = resolve(process.cwd(), '.quello/picks.json')

const empty = (): QuelloPicksFile => ({
  version: 1,
  updatedAt: new Date().toISOString(),
  picks: [],
})

function normalize(input: unknown): QuelloPicksFile {
  const picks = (input as QuelloPicksFile | undefined)?.picks
  const valid = Array.isArray(picks)
    ? picks.filter(
        (p): p is QuelloPick => typeof p?.id === 'number' && typeof p?.selector === 'string',
      )
    : []
  return {
    version: 1,
    updatedAt: new Date().toISOString(),
    picks: [...valid].sort((a, b) => a.id - b.id),
  }
}

export async function GET() {
  try {
    return NextResponse.json(normalize(JSON.parse(await readFile(FILE, 'utf8'))))
  } catch {
    return NextResponse.json(empty())
  }
}

export async function POST(request: Request) {
  const payload = normalize(await request.json())
  await mkdir(dirname(FILE), { recursive: true })
  await writeFile(FILE, `${JSON.stringify(payload, null, 2)}\n`, 'utf8')
  return NextResponse.json({ ok: true, count: payload.picks.length })
}
```

## The agent file

The plugins write it for you; here you write it once, by hand. Copy the `<!-- quello:start -->`
section from [any playground](https://github.com/Linko91/quello/tree/main/playgrounds) into your
`AGENTS.md`, or run `npx @quello/cli --no-serve` once in the project and let it create the file.

## What a pick will know

The component name, always. **Not** the line: Next compiles JSX with SWC, which does not annotate
source locations the way Vite's Babel transform does. Both are React — the difference is the
compiler, not the framework.

```json
{ "framework": { "framework": "react", "component": "OverviewPage" } }
```
